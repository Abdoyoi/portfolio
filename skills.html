Java: linked list removal and duplicate detection (real code from Whispr)

Removing a chat means re-linking its neighbours, including the special cases at the start and end of the list. The contact manager uses a HashSet so it never has to scan every contact to spot a duplicate.

// From Whispr: ChatLinkedList.java (doubly linked list of chats)
public void remove(String chatId) {
    ChatNode current = head;
    while (current != null) {
        if (current.chat.getChatId().equals(chatId)) {
            if (current.prev != null) current.prev.next = current.next;
            else head = current.next;                 // removing the first chat
            if (current.next != null) current.next.prev = current.prev;
            else tail = current.prev;                 // removing the last chat
            size--;
            return;
        }
        current = current.next;
    }
}

Python: scoring Wordle guesses (real code from my search agents)

Every guess splits the words that are still possible into up to 243 groups, one for each colour pattern the game could show. These three functions rank a guess from those group sizes: entropy (information gained in bits, higher is better), expected size (average number of words left, lower is better) and minimax (size of the biggest group, lower is better). The full file is in the team repository.

# From wordle/agents.py: the three heuristics I implemented
def entropy_score(counts: np.ndarray, n_candidates: int) -> np.ndarray:
    """Return information gain, in bits, for each partition row."""
    if n_candidates <= 0:
        raise ValueError("n_candidates must be positive")
    counts = np.asarray(counts, dtype=np.float64)
    probabilities = counts / n_candidates
    with np.errstate(divide="ignore", invalid="ignore"):
        # Empty partitions contribute zero rather than log2(0).
        return -np.sum(
            np.where(probabilities > 0, probabilities * np.log2(probabilities), 0),
            axis=1,
        )


def expected_size_score(counts: np.ndarray, n_candidates: int) -> np.ndarray:
    """Return the expected number of candidates remaining after a guess."""
    if n_candidates <= 0:
        raise ValueError("n_candidates must be positive")
    counts = np.asarray(counts, dtype=np.float64)
    return np.sum(counts * counts, axis=1) / n_candidates


def minimax_score(counts: np.ndarray) -> np.ndarray:
    """Return the size of the largest feedback partition for each guess."""
    return np.max(np.asarray(counts), axis=1)

Python: choosing a guess and filtering the candidates (real code, excerpts)

The agent counts, for every legal guess, how many candidates fall into each feedback group, plays the best-scoring guess for the chosen heuristic, and then keeps only the answers that would have produced the feedback it received. It looks one guess ahead and never backtracks, so it is greedy heuristic search rather than a full search tree.

# From wordle/agents.py (excerpts; input checks and bookkeeping left out)

class Engine:
    def partition_counts(self, cand, guess_ids):
        ...
        # Each row is a histogram of feedback patterns for one possible guess.
        # Indexing P by cand keeps the calculation limited to live answers.
        counts = np.zeros((len(guess_ids), N_PATTERNS), dtype=np.int32)
        for row, guess_id in enumerate(guess_ids):
            counts[row] = np.bincount(
                self.P[guess_id, cand],
                minlength=N_PATTERNS,
            )
        return counts


class SearchAgent:
    def choose(self) -> str:
        ...
        if len(self.cand) == 1:
            # Once one answer remains, guessing it is always optimal.
            guess_id = int(self.engine.answer_guess_ids[self.cand[0]])
        else:
            ...
            counts = self.engine.partition_counts(self.cand, available)
            self.evals += len(available)
            if self.heuristic == "entropy":
                scores = entropy_score(counts, len(self.cand))
                # Higher entropy means a more informative partition.
                guess_id = int(available[int(np.argmax(scores))])
            elif self.heuristic == "expected_size":
                scores = expected_size_score(counts, len(self.cand))
                # Lower expected remainder means faster candidate reduction.
                guess_id = int(available[int(np.argmin(scores))])
            else:
                scores = minimax_score(counts)
                # Minimax limits the largest group left by any feedback.
                guess_id = int(available[int(np.argmin(scores))])
        ...

    def observe(self, guess: str, pattern: int) -> None:
        ...
        guess_id = self.engine.guess_index[guess]
        # Keep only answers whose precomputed feedback matches the observation.
        self.cand = self.cand[self.engine.P[guess_id, self.cand] == int(pattern)]

C: looking up a user in a hash table

Hashing the user ID jumps straight to the right slot instead of checking every user. Collisions are handled with a linked list.

/* Simplified illustration of hash-table lookup (not the project's exact code). */
#include <string.h>

#define TABLE_SIZE 101

typedef struct User {
    char id[32];
    char password_hash[64];
    struct User *next;          /* chaining handles collisions */
} User;

static User *table[TABLE_SIZE];

static unsigned int hash(const char *s) {
    unsigned int h = 5381;
    while (*s) h = h * 33 + (unsigned char)*s++;
    return h % TABLE_SIZE;
}

User *find_user(const char *id) {
    for (User *u = table[hash(id)]; u != NULL; u = u->next)
        if (strcmp(u->id, id) == 0)
            return u;           /* found: average O(1) lookup */
    return NULL;
}
